# AgentGate configuration — copy to .env (or export these) and adjust.
# All values are optional in development; the ones marked REQUIRED are
# enforced at startup when NODE_ENV=production.

# --- Broker service ---
AGENTGATE_BROKER_PORT=4790
# Loopback by default. Only change this if TLS is terminated in front of the
# broker; set AGENTGATE_ALLOW_PUBLIC_BIND=1 to confirm a 0.0.0.0 bind.
AGENTGATE_BROKER_HOST=127.0.0.1

# --- Storage ---
# Where registry, audit, and session state live. Back this up: it holds the
# organisation's root signing key.
AGENTGATE_DATA_DIR=./data

# --- Security ---
# REQUIRED in production. Minimum 32 characters. Guards the /audit endpoints.
# Generate with: openssl rand -hex 32
AGENTGATE_ADMIN_TOKEN=

# How long an issued token is valid. Keep this short — it bounds the damage
# from a leaked credential. Production rejects anything over 1 hour.
AGENTGATE_TOKEN_TTL_MS=900000

# How long an Agent Identity Card is valid before reissue (default 30 days).
AGENTGATE_AGENT_CARD_TTL_MS=2592000000

# Replay window: maximum age of an accepted request, and how long a nonce is
# remembered. Lower is stricter but less tolerant of client clock skew.
AGENTGATE_NONCE_WINDOW_MS=300000

# Maximum accepted HTTP request body.
AGENTGATE_MAX_BODY_BYTES=65536

# --- Logging ---
# debug | info | warn | error
AGENTGATE_LOG_LEVEL=info

# --- Client side (developer machine / agent process) ---

# --- GitHub App (the credential `git push` actually uses) ---
# Unset, the broker issues AgentGate session tokens only — verifiable by
# AgentGate components and meaningless to GitHub. Set, it exchanges each
# authorization decision for an installation token scoped to one repository.
# In production these are all-or-nothing: setting any one requires all of
# them, because a half-configured exchange fails at the first push.
# Requires: npm install --no-save @octokit/auth-app
# AGENTGATE_GITHUB_APP_ID=
# AGENTGATE_GITHUB_INSTALLATION_ID=

# The account the installation belongs to. This is a security control, not a
# convenience: bare repository names resolve against the installation's
# account, so without it a request for "attacker/api" would mint a token
# scoped to "your-org/api".
# AGENTGATE_GITHUB_OWNER=

# Prefer the path form. Environment variables reach process listings and
# crash dumps far more readily than a 0600 file; production refuses to start
# if this file is readable by group or others.
# AGENTGATE_GITHUB_PRIVATE_KEY_PATH=./github-app-key.pem
# AGENTGATE_GITHUB_PRIVATE_KEY=
